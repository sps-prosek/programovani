const uint8_t PIN_LED = 9;

void setup() {
  pinMode(PIN_LED, OUTPUT);
}

void loop() {
  digitalWrite(PIN_LED, HIGH);
  delay(500);
  digitalWrite(PIN_LED, LOW);
  delay(500);
}
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Krátká ukázka a výstup na sériové lince}
  Pro drobné útržky se hodí styl bez číslování řádků; úsek, o kterém se
  zrovna mluví, jde zvýraznit makrem \kod{\textbackslash zvyrazni}:
\begin{lstlisting}[style=arduino_bez_cisel]
for (int i = 0; i < 3; i++) {
  Serial.println((*@\zvyrazni{i}@*));
}
\end{lstlisting}

  Co se objeví v sériovém monitoru:
\begin{vystup}
0
1
2
\end{vystup}
\end{frame}
